#+title: DriftWM Configuration

* DriftWM Configuration
Trackpad-first infinite canvas compositor via [[https://github.com/malbiruk/driftwm][malbiruk/driftwm]].

** System Configuration
Loaded via flake's selectedDesktop when userSettings.wm = "driftwm".

Base system config (greetd, uwsm, seatd, dbus, PAM):
#+begin_src nix :tangle ./modules/driftwm/system.nix
{ config, pkgs, inputs, ... }:

{
  imports = [ ../../system/wayland.nix ];

  services.xserver.enable = true;
  services.greetd = {
    enable = true;
    settings = {
      default_session = {
        command = "${pkgs.greetd.tuigreet}/bin/tuigreet --time --remember --remember-session --cmd 'uwsm start driftwm'";
        user = "greeter";
      };
    };
  };
  services.xserver.displayManager.lightdm.enable = false;
  programs.uwsm = {
    enable = true;
    waylandCompositors = {
      driftwm = {
        prettyName = "driftwm";
        comment = "driftwm Wayland compositor managed by UWSM";
        binPath = "${pkgs.writeShellScript "driftwm-uwsm-wrapper" "exec ${inputs.driftwm.packages.${pkgs.stdenv.system}.default}/bin/driftwm start"}";
      };
    };
  };
  services.seatd.enable = true;
  services.dbus.enable = true;

  # PAM services for screen lockers
  security.pam.services.swaylock = {};
  security.pam.services.hyprlock = {};

  xdg.portal = {
    enable = true;
    extraPortals = [ pkgs.xdg-desktop-portal-wlr ];
    configPackages = [ inputs.driftwm.packages.${pkgs.stdenv.system}.default ];
  };

  services.upower.enable = true;
  services.power-profiles-daemon.enable = true;
}
#+end_src

** Home Configuration
DriftWM + Noctalia shell. Replaces waybar, swaync, swayosd, and custom widgets with Noctalia's built-in bar, launcher, notifications, and OSD.

The full home configuration is generated from this file.
:PROPERTIES:
:header-args:nix: :noweb-ref driftwm-home
:END:
#+begin_src nix :tangle ./modules/driftwm/home.nix :noweb yes
{ config, pkgs, inputs, lib, ... }:

let
  driftwmPkg = inputs.driftwm.packages.${pkgs.stdenv.system}.default;
  driftwmConfig = pkgs.writeText "driftwm-config.toml" ''
    <<driftwm-config-toml>>
  '';
in
{
  <<driftwm-home.imports>>
  <<driftwm-home.activation>>
  <<driftwm-home.packages>>
  <<driftwm-home.files>>
  <<driftwm-home.systemd>>
  <<driftwm-home.gpg>>
}
#+end_src

*** Activation Scripts
#+begin_src nix :noweb-ref driftwm-home.activation
  home.activation.checkDriftwmConfig = lib.hm.dag.entryBefore ["copyDriftwmConfig"] ''
    ${driftwmPkg}/bin/driftwm --config ${driftwmConfig} --check-config || true
  '';

  home.activation.copyDriftwmConfig = lib.hm.dag.entryAfter ["writeBoundary"] ''
    if [ ! -f "${config.home.homeDirectory}/.config/driftwm/config.toml" ]; then
      mkdir -p "${config.home.homeDirectory}/.config/driftwm"
      install -m644 "${driftwmConfig}" "${config.home.homeDirectory}/.config/driftwm/config.toml"
    fi
  '';

** Noctalia Shell
The Noctalia shell component is imported as a reusable module from [[../../modules/noctalia][modules/noctalia/]].
DriftWM provides its own =noctalia.json= settings file.

** Noctalia Settings Sync
Noctalia's settings are edited via its graphical interface. Run this script to sync them back into the repo:
#+begin_src sh :tangle ./modules/driftwm/sync-noctalia.sh
#!/bin/sh
SRC="$HOME/.config/noctalia/settings.json"
DST="$HOME/.dotfiles/modules/driftwm/noctalia.json"
cp "$SRC" "$DST"
echo "noctalia.json synced from GUI"
#+end_src
